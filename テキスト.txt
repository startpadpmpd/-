    <script>
        /* Web Audio API Sound Engine */
        let audioCtx = null;
        let soundEnabled = true;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }

        function playSound(type) {
            if (!soundEnabled) return;
            initAudio();
            if (!audioCtx) return;

            const now = audioCtx.currentTime;
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);

            if (type === 'correct') {
                // 正解のチャイム (ピンポン)
                osc.type = 'sine';
                osc.frequency.setValueAtTime(523.25, now); // C5
                osc.frequency.setValueAtTime(659.25, now + 0.12); // E5
                gain.gain.setValueAtTime(0.2, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.5);
                osc.start(now);
                osc.stop(now + 0.5);
            } else if (type === 'wrong') {
                // 失敗音 (ブッ)
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(150, now);
                osc.frequency.setValueAtTime(120, now + 0.1);
                gain.gain.setValueAtTime(0.2, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.3);
                osc.start(now);
                osc.stop(now + 0.3);
            } else if (type === 'click') {
                // タップ音
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(400, now);
                gain.gain.setValueAtTime(0.1, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.05);
                osc.start(now);
                osc.stop(now + 0.05);
            } else if (type === 'clear') {
                // ファンファーレ（全問クリア）
                const notes = [523.25, 659.25, 783.99, 1046.50];
                notes.forEach((freq, idx) => {
                    const o = audioCtx.createOscillator();
                    const g = audioCtx.createGain();
                    o.connect(g);
                    g.connect(audioCtx.destination);
                    o.frequency.value = freq;
                    g.gain.setValueAtTime(0.15, now + idx * 0.1);
                    g.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.1 + 0.3);
                    o.start(now + idx * 0.1);
                    o.stop(now + idx * 0.1 + 0.3);
                });
            }
        }

        }

        // ランダム問題の作成
        function generateNewProblem() {
            playSound('click');
            let div, dor;

            // レベルごとの数値生成
            if (currentLevel === 1) { // 2桁÷1桁
                dor = Math.floor(Math.random() * 8) + 2; // 2~9
                if (allowRemainder) {
                    div = Math.floor(Math.random() * 80) + 15; // 15~94
                } else {
                    const q = Math.floor(Math.random() * 25) + 10;
                    div = dor * q;
                    if (div > 99) div = dor * Math.floor(99 / dor);
                }
            } else if (currentLevel === 2) { // 3桁÷1桁
                dor = Math.floor(Math.random() * 8) + 2;
                if (allowRemainder) {
                    div = Math.floor(Math.random() * 800) + 100; // 100~899
                } else {
                    const q = Math.floor(Math.random() * 200) + 50;
                    div = dor * q;
                    if (div > 999) div = dor * Math.floor(999 / dor);
                }
            } else if (currentLevel === 3) { // 2桁÷2桁
                dor = Math.floor(Math.random() * 80) + 11; // 11~90
                const maxQ = Math.floor(99 / dor);
                const q = Math.max(1, Math.floor(Math.random() * maxQ) + 1);
                if (allowRemainder) {
                    const rem = Math.floor(Math.random() * dor);
                    div = Math.min(99, dor * q + rem);
                } else {
                    div = dor * q;
                }
            } else { // 3桁÷2桁
                dor = Math.floor(Math.random() * 80) + 12; // 12~91
                if (allowRemainder) {
                    div = Math.floor(Math.random() * 800) + 150; // 150~949
                } else {
                    const q = Math.floor(Math.random() * 40) + 10;
                    div = dor * q;
                    if (div > 999) div = dor * Math.floor(999 / dor);
                }
            }

            // 簡単すぎる問題の回避
            if (div < dor) div = dor + Math.floor(Math.random() * 20) + 5;

            dividend = div;
            divisor = dor;

            initProblem();
        }

        function startCustomProblem() {
            const divInput = parseInt(document.getElementById('custom-dividend').value);
            const dorInput = parseInt(document.getElementById('custom-divisor').value);

            if (isNaN(divInput) || isNaN(dorInput) || dorInput <= 0 || divInput < dorInput) {
                alert('正しく「わられる数」と「わる数（わられる数より小さい数）」を入力してね！');
                return;
            }

            dividend = divInput;
            divisor = dorInput;
            initProblem();
        }

        // 筆算の全ステップ解法を計算構築
        function solveHissan(div, dor) {
            const divStr = div.toString();
            const dorStr = dor.toString();

            const numDivDigits = divStr.length;
            const numDorDigits = dorStr.length;

            let stepList = [];
            let currentVal = 0;
            let quotientStr = "";

            // グリッドサイズの計算
            // 列数: わる数桁 + 1 + わられる数桁 + 1
            const cols = numDorDigits + 1 + numDivDigits + 1;
            
            // 各パートの計算
            let gridMap = {}; // (row, col) -> { val, type, stepIndex }
            
            // 行の割り当て
            // Row 0: 商 (Quotient)
            // Row 1: わられる数 & わる数 (Dividend & Divisor)
            
            let divStartCol = numDorDigits + 1; // わられる数の開始列
            let dorStartCol = 0; // わる数の開始列

            // Step構築
            let currentGridRow = 2; // かけ算結果を書く最初の行
            
            let tempValStr = "";
            let processedDigits = 0;

            for (let i = 0; i < numDivDigits; i++) {
                tempValStr += divStr[i];
                let curVal = parseInt(tempValStr);
                let colPos = divStartCol + i; // 現在処理中の桁の列

                if (curVal < dor) {
                    // 商がたたない場合（ただし先頭でない場合、または最後までいったら0）
                    if (quotientStr.length > 0) {
                        quotientStr += "0";
                        stepList.push({
                            type: 'tateru',
                            val: 0,
                            row: 0,
                            col: colPos,
                            hint: `${curVal}の中に${dor}は入らないので「0」をたてるよ。`,
                            targetCellId: `r0c${colPos}`
                        });
                    }
                    continue;
                }

                // ① たてる
                let qDigit = Math.floor(curVal / dor);
                quotientStr += qDigit.toString();
                
                stepList.push({
                    type: 'tateru',
                    val: qDigit,
                    row: 0,
                    col: colPos,
                    hint: `${curVal} の中に ${dor} は何回はいるかな？ 商は「${qDigit}」をたてるよ！`,
                    targetCellId: `r0c${colPos}`
                });

                // ② かける
                let prod = qDigit * dor;
                let prodStr = prod.toString();
                let prodStartCol = colPos - prodStr.length + 1;
                
                let prodCellIds = [];
                for (let p = 0; p < prodStr.length; p++) {
                    let pCol = prodStartCol + p;
                    let pVal = parseInt(prodStr[p]);
                    prodCellIds.push({ row: currentGridRow, col: pCol, val: pVal, id: `r${currentGridRow}c${pCol}` });
                }

                stepList.push({
                    type: 'kakeru',
                    vals: prodCellIds,
                    hint: `たてた「${qDigit}」と わる数「${dor}」をかけるよ。 ${qDigit} × ${dor} = ${prod}`,
                });

                // ③ ひく
                let rem = curVal - prod;
                let remStr = rem.toString();
                let remStartCol = colPos - remStr.length + 1;
                currentGridRow++; // 引き算結果の行

                let remCellIds = [];
                for (let r = 0; r < remStr.length; r++) {
                    let rCol = remStartCol + r;
                    let rVal = parseInt(remStr[r]);
                    remCellIds.push({ row: currentGridRow, col: rCol, val: rVal, id: `r${currentGridRow}c${rCol}` });
                }

                stepList.push({
                    type: 'hiku',
                    vals: remCellIds,
                    lineAtRow: currentGridRow - 1, // 引き算の横線を表示する行
                    lineStartCol: prodStartCol,
                    lineEndCol: colPos,
                    hint: `ひき算をするよ。 ${curVal} - ${prod} = ${rem}`,
                });

                // 次の桁をおろす準備
                if (i < numDivDigits - 1) {
                    let nextDigit = divStr[i + 1];
                    let nextCol = divStartCol + i + 1;
                    
                    stepList.push({
                        type: 'orosu',
                        val: parseInt(nextDigit),
                        row: currentGridRow,
                        col: nextCol,
                        hint: `つぎの桁の「${nextDigit}」をまっすぐ下におろしてこよう！`,
                        targetCellId: `r${currentGridRow}c${nextCol}`
                    });

                    tempValStr = rem.toString(); // あまり + 次のおろした数字（ループで追加される）
                    currentGridRow++;
                } else {
                    // 終了
                }
            }

            return {
                steps: stepList,
                numRows: currentGridRow + 1,
                numCols: cols,
                divStartCol: divStartCol,
                dorStartCol: dorStartCol,
                finalQuotient: Math.floor(div / dor),
                finalRemainder: div % dor
            };
        }

        // 筆算ボードの描画初期化
        function initProblem() {
            userInputs = {};
            currentStepIndex = 0;

            const solution = solveHissan(dividend, divisor);
            steps = solution.steps;

            // 数式表示のアップデート
            document.getElementById('problem-expression').innerText = `${dividend} ÷ ${divisor} = ?`;

            const board = document.getElementById('hissan-board');
            board.innerHTML = '';

            const numRows = solution.numRows;
            const numCols = solution.numCols;

            board.style.gridTemplateColumns = `repeat(${numCols}, minmax(0, 1fr))`;

            // 空のグリッドを作成
            for (let r = 0; r < numRows; r++) {
                for (let c = 0; c < numCols; c++) {
                    const cell = document.createElement('div');
                    cell.id = `cell-r${r}c${c}`;
                    cell.className = 'cell';

                    // わる数の配置 (Row 1, dorStartCol...)
                    const dorStr = divisor.toString();
                    if (r === 1 && c >= solution.dorStartCol && c < solution.dorStartCol + dorStr.length) {
                        cell.innerText = dorStr[c - solution.dorStartCol];
                        cell.classList.add('cell-fixed');
                    }

                    // わられる数の配置 (Row 1, divStartCol...)
                    const divStr = dividend.toString();
                    if (r === 1 && c >= solution.divStartCol && c < solution.divStartCol + divStr.length) {
                        cell.innerText = divStr[c - solution.divStartCol];
                        cell.classList.add('cell-fixed');
                        cell.classList.add('hissan-border-top'); // 上線（割る記号の横棒）
                        if (c === solution.divStartCol) {
                            cell.classList.add('hissan-border-left'); // 左カーブ線
                        }
                    }

                    board.appendChild(cell);
                }
            }

            updateStepUI();
        }

        // 現在のステップに基づいてUI（アクティブセル、ヒント、合言葉）を更新
        function updateStepUI() {
            // すべての入力セルのアクティブ状態解除
            document.querySelectorAll('.cell').forEach(cell => {
                cell.classList.remove('active');
            });

            if (currentStepIndex >= steps.length) {
                // すべて完了！
                onProblemComplete();
                return;
            }

            const step = steps[currentStepIndex];
            
            // 合言葉バッジの更新
            updateBadgeUI(step.type);

            // ヒント文の更新
            const hintText = document.getElementById('hint-text');
            if (currentMode === 'guide') {
                hintText.innerText = step.hint;
            } else {
                hintText.innerText = 'つぎのマスの数字をかんがえて入力しよう！';
            }

            // 引き算の横線の描画 (もしあれば)
            if (step.type === 'hiku' && step.lineAtRow) {
                for (let c = step.lineStartCol; c <= step.lineEndCol; c++) {
                    const lineCell = document.getElementById(`cell-r${step.lineAtRow}c${c}`);
                    if (lineCell) lineCell.classList.add('hissan-line');
                }
            }

            // アクティブな入力ターゲットの設定
            let targetIds = [];
            if (step.type === 'tateru' || step.type === 'orosu') {
                targetIds.push(step.targetCellId);
            } else if (step.type === 'kakeru' || step.type === 'hiku') {
                step.vals.forEach(v => targetIds.push(v.id));
            }

            targetIds.forEach(id => {
                const cell = document.getElementById(`cell-${id}`);
                if (cell) {
                    cell.classList.add('cell-input', 'active');
                    if (userInputs[id] !== undefined) {
                        cell.innerText = userInputs[id];
                    } else {
                        cell.innerText = '';
                    }
                }
            });

            activeCellId = targetIds[0]; // 単一のフォーカス対象
            
            document.getElementById('status-bar').className = "w-full text-center py-2 px-4 rounded-xl bg-blue-100 text-blue-800 font-bold text-sm min-h-[40px] flex items-center justify-center";
            document.getElementById('status-bar').innerText = "数字を選んで「けってい」を押してね！";
        }

        // バッジのハイライト
        function updateBadgeUI(stepType) {
            ['tateru', 'kakeru', 'hiku', 'orosu'].forEach(t => {
                const badge = document.getElementById(`step-badge-${t}`);
                if (t === stepType) {
                    badge.className = 'step-badge bg-amber-400 text-amber-950 py-1.5 rounded-lg border-2 border-amber-500 shadow scale-105 font-black';
                } else {
                    badge.className = 'step-badge bg-slate-100 text-slate-400 py-1.5 rounded-lg border';
                }
            });
        }

        // テンキー入力処理
        function pressKey(key) {
            if (currentStepIndex >= steps.length) return;

            const step = steps[currentStepIndex];

            if (key === 'backspace') {
                playSound('click');
                if (activeCellId) {
                    delete userInputs[activeCellId];
                    const cell = document.getElementById(`cell-${activeCellId}`);
                    if (cell) cell.innerText = '';
                }
                return;
            }

            if (key === 'enter') {
                verifyStep();
                return;
            }

            // 数字キー入力
            if (key >= '0' && key <= '9') {
                playSound('click');
                // 複数入力ターゲットがある場合、現在空のマスに入力
                let targetIds = [];
                if (step.type === 'tateru' || step.type === 'orosu') {
                    targetIds.push(step.targetCellId);
                } else if (step.type === 'kakeru' || step.type === 'hiku') {
                    step.vals.forEach(v => targetIds.push(v.id));
                }

                // 選択されているマスまたは最初の未入力マスへ
                let fillId = targetIds.find(id => userInputs[id] === undefined) || targetIds[targetIds.length - 1];
                userInputs[fillId] = parseInt(key);
                
                const cell = document.getElementById(`cell-${fillId}`);
                if (cell) cell.innerText = key;
                
                // もし次の未入力マスがあればフォーカス移動
                let nextEmpty = targetIds.find(id => userInputs[id] === undefined);
                if (nextEmpty) {
                    activeCellId = nextEmpty;
                }
            }
        }

        // 入力答え合わせ
        function verifyStep() {
            const step = steps[currentStepIndex];
            let isCorrect = true;

            if (step.type === 'tateru' || step.type === 'orosu') {
                const inputVal = userInputs[step.targetCellId];
                if (inputVal !== step.val) {
                    isCorrect = false;
                }
            } else if (step.type === 'kakeru' || step.type === 'hiku') {
                step.vals.forEach(v => {
                    if (userInputs[v.id] !== v.val) {
                        isCorrect = false;
                    }
                });
            }

            if (isCorrect) {
                // 正解！
                playSound('correct');
                
                // マスを正解デザインにして固定
                let targetIds = [];
                if (step.type === 'tateru' || step.type === 'orosu') targetIds.push(step.targetCellId);
                else step.vals.forEach(v => targetIds.push(v.id));

                targetIds.forEach(id => {
                    const cell = document.getElementById(`cell-${id}`);
                    if (cell) {
                        cell.classList.remove('active', 'cell-input');
                        cell.classList.add('cell-fixed', 'text-emerald-600');
                    }
                });

                currentStepIndex++;
                updateStepUI();

            } else {
                // 不正解
                playSound('wrong');
                document.getElementById('status-bar').className = "w-full text-center py-2 px-4 rounded-xl bg-rose-100 text-rose-800 font-bold text-sm min-h-[40px] flex items-center justify-center animate-bounce";
                document.getElementById('status-bar').innerText = "おしい！もういちど数字をたしかめてみよう！";

                let targetIds = [];
                if (step.type === 'tateru' || step.type === 'orosu') targetIds.push(step.targetCellId);
                else step.vals.forEach(v => targetIds.push(v.id));

                targetIds.forEach(id => {
                    const cell = document.getElementById(`cell-${id}`);
                    if (cell) {
                        cell.classList.add('error');
                        setTimeout(() => cell.classList.remove('error'), 500);
                    }
                });
            }
        }

        // 問題クリア時の処理
        function onProblemComplete() {
            playSound('clear');
            score++;
            document.getElementById('score-count').innerText = score;

            const rem = dividend % divisor;
            const q = Math.floor(dividend / divisor);
            const ansStr = rem === 0 ? `商: ${q}` : `商: ${q} あまり ${rem}`;

            document.getElementById('status-bar').className = "w-full text-center py-2 px-4 rounded-xl bg-emerald-500 text-white font-extrabold text-base min-h-[40px] flex items-center justify-center shadow-lg";
            document.getElementById('status-bar').innerText = `🎉 だいせいかい！ (${ansStr})`;

            document.getElementById('hint-text').innerText = "すごい！かんぺきに解けたね！「つぎの問題」ボタンを押してチャレンジしよう！";
            
            // バッジのハイライト全解除
            ['tateru', 'kakeru', 'hiku', 'orosu'].forEach(t => {
                const badge = document.getElementById(`step-badge-${t}`);
                badge.className = 'step-badge bg-emerald-100 text-emerald-700 font-bold py-1.5 rounded-lg border border-emerald-300';
            });
        }
